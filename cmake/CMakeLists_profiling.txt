# PA02 Step 1: 병목 확인 (PA01 final score_all + fast_matcher_profiled)
# cp cmake/CMakeLists_profiling.txt CMakeLists.txt && catkin_make -DCMAKE_BUILD_TYPE=Release

cmake_minimum_required(VERSION 3.8)
project(cartographer_parallel LANGUAGES CXX CUDA)

find_package(catkin REQUIRED COMPONENTS
  roscpp std_msgs sensor_msgs geometry_msgs nav_msgs visualization_msgs)

catkin_package(
  INCLUDE_DIRS include
  LIBRARIES assignment_cpu_lib fast_matcher_lib
  CATKIN_DEPENDS roscpp std_msgs sensor_msgs geometry_msgs nav_msgs visualization_msgs)

link_directories(/usr/local/cuda/lib64)
include_directories(include ${catkin_INCLUDE_DIRS})
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 14)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} -gencode arch=compute_53,code=sm_53")

# score_all: PA01 최종 (gpu_final + scale_1024)
add_library(assignment_cpu_lib src/score_all_pa02_final.cpp)
set_source_files_properties(src/score_all_pa02_final.cpp PROPERTIES LANGUAGE CUDA)
target_link_libraries(assignment_cpu_lib ${catkin_LIBRARIES} cudart)

# fast_matcher: 타이머 삽입 버전 (함수별 시간 측정)
add_library(fast_matcher_lib src/fast_matcher_profiled.cpp)
target_link_libraries(fast_matcher_lib assignment_cpu_lib ${catkin_LIBRARIES})

add_executable(fast_correlative_node src/fast_correlative_node.cpp)
target_link_libraries(fast_correlative_node fast_matcher_lib ${catkin_LIBRARIES})

install(TARGETS assignment_cpu_lib fast_matcher_lib fast_correlative_node
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch maps bags DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
